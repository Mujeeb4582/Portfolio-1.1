   import { useState } from 'react'

   import { Button } from '@/app/ui/button'
   import { UnderLine } from '@/app/ui/underLine'
   import { cn } from '@/app/lib/utils'

3. `BrowserFrame` component (accepts `{ title: string; screenshotHeight: number; screenshotPath: string; projectTitle: string }` props):
   - Outer: `<div className="rounded-t-lg overflow-hidden">`

   - URL bar: `<div className="ml-2 flex-1 rounded bg-background/50 px-2 py-1 text-[10px] text-muted-foreground font-mono truncate">` containing title text
   - Screenshot container: `<div className="relative overflow-hidden group" style={{ height: screenshotHeight }}>` with `<Image>` fill + objectFit cover + objectPosition top + loading lazy + alt text `{projectTitle} — web application screenshot`
   - Placeholder swap: accepts `showPlaceholder` and `onImageError` props

4. `PhoneFrame` component (accepts same props pattern):
   - Outer: `<div className="mx-auto max-w-[220px] rounded-[2rem] border-4 border-muted bg-muted p-1 shadow-inner">`

   - Role + company: `<p className="font-jetbrains text-code-m text-muted-foreground">{project.role} at {project.company}</p>`
   - Description: `<p className="font-jetbrains text-para-m text-foreground line-clamp-3">{project.description}</p>`
   - Tech pills: `<div className="flex flex-wrap gap-2 mt-2">` with pills using `<span className="font-jetbrains rounded-full bg-muted px-2 py-1 text-code-m text-muted-foreground">`

- Do NOT set `priority` on any project Image — all are below fold, use `loading="lazy"`
- For the featured card, the screenshot height differs on mobile vs desktop.
