```tsx
import { render, screen } from '@testing-library/react'
import ProjectsSection from '@/app/ui/homepage/projectsSection'

describe('ProjectsSection', () => {
  it('PROJ-01: renders section heading "Projects"', () => {
    render(<ProjectsSection />)
    expect(screen.getByRole('heading', { name: 'Projects', level: 2 })).toBeInTheDocument()
  })

  it('PROJ-04: renders all 6 project titles', () => {
    render(<ProjectsSection />)
    expect(screen.getByText('Buildable')).toBeInTheDocument()
    expect(screen.getByText('MISA App')).toBeInTheDocument()
    expect(screen.getByText('Uber-like App')).toBeInTheDocument()
    expect(screen.getByText('Re-View')).toBeInTheDocument()
    expect(screen.getByText('LSTN')).toBeInTheDocument()
    expect(screen.getByText('WellShared')).toBeInTheDocument()
  })

  it('PROJ-01: renders tech stack pills for Buildable', () => {
    render(<ProjectsSection />)
    expect(screen.getByText('Next.js')).toBeInTheDocument()
    expect(screen.getByText('TypeScript')).toBeInTheDocument()
  })

  it('PROJ-05: renders BrowserFrame decoration for web projects', () => {
    const { container } = render(<ProjectsSection />)
    // BrowserFrame has a browser bar with traffic light dots
    // The URL bar div contains font-mono text class
    const monoBars = container.querySelectorAll('.font-mono.truncate')
    expect(monoBars.length).toBeGreaterThan(0)
  })

  it('PROJ-05: renders PhoneFrame decoration for mobile projects', () => {
    const { container } = render(<ProjectsSection />)
    // PhoneFrame has max-w-[220px] outer div
    // Use a selector that targets the phone bezel outer wrapper
    const phoneFrames = container.querySelectorAll('[class*="max-w-\\[220px\\]"]')
    expect(phoneFrames.length).toBeGreaterThanOrEqual(2) // MISA App + Uber-like App
  })

  it('PROJ-01: renders role + company meta for each project', () => {
    render(<ProjectsSection />)
    expect(screen.getByText('Full-Stack Web Developer at Wonder Crafts')).toBeInTheDocument()
    expect(screen.getByText('React Native Developer at RGX Labs')).toBeInTheDocument()
  })
})
```
